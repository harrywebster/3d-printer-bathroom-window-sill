# Build dependencies for src/build.py. Pinned, because the mesh booleans and
# the convex hulls must give the same numbers on every machine — the README
# quotes volumes and the checks compare them.
#
# Install into a venv:  make deps      (uv pip, or: pip install -r requirements.txt)
# Verified on CPython 3.13, Linux x86_64.

# --- geometry ---
trimesh==5.1.0
manifold3d==3.5.3        # the boolean engine every difference/union names
shapely==2.1.2           # 2-D outlines the solids are built from
numpy==2.5.3

# --- drawing ---
matplotlib==3.11.2       # views + schematic, Agg backend only

# --- trimesh's optional backends, all of them load-bearing here ---
# Not optional in practice: without these the build dies part-way through,
# after it has already written some of the deliverables.
scipy==1.18.1            # convex_hull -> QHull. Every ribbed/sloped hull needs it.
lxml==6.1.3              # reloading the shipped 3MF for the volume check
rtree==1.4.1             # spatial index for the proximity queries
networkx==3.6.1          # mesh graph ops behind body_count
